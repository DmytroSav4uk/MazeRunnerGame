import {Component, Inject} from '@angular/core';

import {
  MAT_DIALOG_DATA,
  MatDialogActions,
  MatDialogContent,
  MatDialogRef,
  MatDialogTitle
} from '@angular/material/dialog';

import {MatButton} from '@angular/material/button';
import {Router} from '@angular/router';
import {MainChar} from '../../../interfaces/mainChar';
import {IItem} from '../../../interfaces/Item';

@Component({
  selector: 'app-inventory-dialog',
  imports: [
    MatDialogContent,
    MatDialogActions,
    MatButton,
    MatDialogTitle
  ],
  templateUrl: './inventory-dialog.html',
  styleUrl: './inventory-dialog.css'
})
export class InventoryDialog {
  items = MainChar.inventory.items;

  constructor(
    public dialogRef: MatDialogRef<InventoryDialog>,
    @Inject(MAT_DIALOG_DATA) public data: any,
    private router: Router
  ) {}

  useItem(item: IItem) {

    const index = MainChar.inventory.items.findIndex(slot => slot.item.name === item.name);
    if (index !== -1) {
      MainChar.inventory.useItem(index, MainChar);
    }
  }

  close() {
    this.dialogRef.close();
  }


  canBeUsed(item: IItem): boolean {
    const currentRoute = this.router.url.toLowerCase();

    if (currentRoute.includes('maze')) {
      return !item.usableInMaze;
    } else if (currentRoute.includes('battle')) {
      return !item.usableInBattle;
    }

    return true;
  }




}

<h2 mat-dialog-title style="text-align: center">Inventory</h2>

<mat-dialog-content class="inventory-content">

  @if (items.length === 0) {
    <div class="empty">
      Your inventory is empty.
    </div>
  }

  @for (stack of items; track stack.item.name) {
    <div class="item-row">
      <div class="item-info">
        <span class="item-name">{{ stack.item.name }}</span>

        @if (stack.quantity > 1) {
          <span class="item-qty">×{{ stack.quantity }}</span>
        }

        <p class="item-desc">{{ stack.item.description }}</p>
      </div>

      <button
        mat-raised-button
        color="primary"
        (click)="useItem(stack.item)"
        [disabled]="canBeUsed(stack.item)"
      >
        Use
      </button>
    </div>
  }


</mat-dialog-content>
